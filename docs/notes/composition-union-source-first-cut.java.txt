// FIRST CUT (2026-09-14) of the query-side union builder with the members' navigations LIFTED
// from their own steps — the capability the STACK needs (docs/LEGACY_ROUTES_AS_COMPOSITION_2026_09_13.md §9).
// Reverted the same day after 337 LOST rows (incomplete, not wrong: no inverse-association
// ends, no per-arm identity, aggregation-aware/~func members untested). Kept as the starting
// point for the 'demand flows into the arms' slice. It replaced ClassSources.unionSourceOf and
// used a NavRewrites helper (readPath / collectReads / renameReads / renameVar) that is now
// NavigateChecker's private walkers.

    /**
     * THE UNION SOURCE — one builder for every union the query side reads:
     * a union as a STACK (the members' functions concatenated; no keys), and
     * a routed navigate's union (one arm per route; the union-row keys the
     * routes' conditions read). Each arm is a LEAF member's own pipeline (a
     * member that is itself a union flattens to its leaves), projecting the
     * class's scalar properties by the member's bindings (a typed NULL where
     * a member lacks one) plus the keys — its own route's reads, NULL for
     * the rest. Then THE UNION'S OWN NAVIGATIONS: for a class-typed
     * property every leaf's function navigates, ONE navigate step above the
     * stack composed from the leaves' own steps — its target the union of
     * their routes, its condition the OR of their conditions re-pointed at
     * columns each arm projects from its own row. Built from nothing but
     * the members' functions: no mapping, no navigator, no key publication.
     * {@code armPipes} is empty (each member's own pipeline) or holds one
     * pipeline per member (a route's mids on top of the member's).
     */
    private ClassSource unionSourceOf(String mappingFqn, String classFqn,
            List<ClassSource> members, List<TypedSpec> armPipes,
            List<com.legend.compiler.spec.typed.TypedNavigate.Route> routes) {
        var cls = ctx.findClass(classFqn).orElseThrow(() ->
                new IllegalStateException("resolver bug: union class '"
                        + classFqn + "' unknown to the model"));
        var one = com.legend.compiler.element.type.Multiplicity.Bounded.ONE;
        var optional = com.legend.compiler.element.type.Multiplicity.Bounded.ZERO_ONE;
        var many = com.legend.compiler.element.type.Multiplicity.Bounded.ZERO_MANY;
        // flatten: leaves, each with the route (if any) of the member it came from
        List<ClassSource> leaves = new ArrayList<>();
        List<com.legend.compiler.spec.typed.TypedNavigate.Route> leafRoute = new ArrayList<>();
        List<TypedSpec> pipes = new ArrayList<>();
        for (int i = 0; i < members.size(); i++) {
            ClassSource m = members.get(i);
            List<ClassSource> ls = unionLeaves.get(m);
            var r = routes.isEmpty() ? null : routes.get(i);
            for (ClassSource leaf : ls != null ? ls : List.of(m)) {
                leaves.add(leaf);
                leafRoute.add(r);
                TypedSpec base = ls == null && !armPipes.isEmpty() ? armPipes.get(i)
                        : Pipelines.materialize(leaf.pipeline(), java.util.Set.of(), classFqn).pipeline();
                pipes.add(ls != null && r != null ? rebaseRows(r.rows(), base) : base);
            }
        }
        List<Type.Column> cols = new ArrayList<>();
        List<com.legend.compiler.element.Property> classTyped = new ArrayList<>();
        for (var p : cls.properties()) {
            if (Type.asClassType(p.type()) instanceof Type.ClassType) {
                classTyped.add(p);
            } else {
                cols.add(new Type.Column(p.name(), p.type(), p.multiplicity()));
            }
        }
        // the union-row keys the ROUTES read, in route order, once per name
        List<Type.Column> keyCols = new ArrayList<>();
        java.util.Set<String> seenKeys = new LinkedHashSet<>();
        for (int i = 0; i < leaves.size(); i++) {
            var r = leafRoute.get(i);
            if (r == null) {
                continue;
            }
            Type.RelationType mRow = Type.requireRelationSchema(pipes.get(i).info().type());
            for (int k = 0; k < r.keyNames().size(); k++) {
                String read = r.targetReads().get(k);
                if (seenKeys.add(r.keyNames().get(k))) {
                    Type kt = pathType(mRow, read);
                    if (kt == null) {
                        throw new MappingResolutionException("route condition reads '" + read
                                + "', which the target set's rows do not carry"
                                + " (class '" + classFqn + "', mapping '" + mappingFqn + "')", classFqn);
                    }
                    keyCols.add(new Type.Column(r.keyNames().get(k), kt, optional));
                }
            }
        }
        // THE UNION'S OWN NAVIGATIONS: per class-typed property every leaf
        // navigates, the leaves' steps composed into one step above the stack
        List<Lift> lifts = new ArrayList<>();
        for (var p : classTyped) {
            Lift lift = liftOf(p, leaves, pipes);
            if (lift != null) {
                lifts.add(lift);
            }
        }
        List<Type.Column> allCols = new ArrayList<>(cols);
        allCols.addAll(keyCols);
        for (Lift l : lifts) {
            allCols.addAll(l.srcCols());
        }
        Type.RelationType rowType = new Type.RelationType(allCols);
        TypedSpec union = null;
        for (int i = 0; i < leaves.size(); i++) {
            ClassSource m = leaves.get(i);
            var r = leafRoute.get(i);
            TypedSpec pipe = pipes.get(i);
            Type.RelationType mRow = Type.requireRelationSchema(pipe.info().type());
            var mInfo = new ExprType(mRow, one);
            List<com.legend.compiler.spec.typed.TypedFuncCol> pcols = new ArrayList<>();
            for (Type.Column c : cols) {
                TypedSpec bnd = m.bindings().get(c.name());
                TypedSpec v = bnd != null ? bnd
                        : new TypedCollection(List.of(), new ExprType(c.type(), optional));
                pcols.add(mixedCol(c.name(), v, mRow, m.rowVar()));
            }
            for (Type.Column kc : keyCols) {
                int k = r == null ? -1 : r.keyNames().indexOf(kc.name());
                TypedSpec v = k >= 0
                        ? pathRead(new TypedVariable(m.rowVar(), mInfo), mRow,
                                r.targetReads().get(k), new ExprType(kc.type(), optional))
                        : new TypedCollection(List.of(), new ExprType(kc.type(), optional));
                pcols.add(mixedCol(kc.name(), v, mRow, m.rowVar()));
            }
            for (Lift l : lifts) {
                for (Type.Column sc : l.srcCols()) {
                    String path = l.srcPathOf(i, sc.name());
                    TypedSpec v = path != null
                            ? pathRead(new TypedVariable(m.rowVar(), mInfo), mRow, path,
                                    new ExprType(sc.type(), optional))
                            : new TypedCollection(List.of(), new ExprType(sc.type(), optional));
                    pcols.add(mixedCol(sc.name(), v, mRow, m.rowVar()));
                }
            }
            TypedSpec arm = new com.legend.compiler.spec.typed.TypedProject(
                    pipe, pcols, new ExprType(Type.relation(rowType), many));
            union = union == null ? arm
                    : new com.legend.compiler.spec.typed.TypedConcatenate(
                            union, arm, new ExprType(Type.relation(rowType), many));
        }
        TypedSpec pipeline = java.util.Objects.requireNonNull(union, "a union with no members");
        List<Type.Column> withSlots = new ArrayList<>(allCols);
        String rowVar = "u_row";
        for (Lift l : lifts) {
            Type.RelationType srcRow = new Type.RelationType(withSlots);
            withSlots.add(new Type.Column(l.property(), l.targetType(), one));
            pipeline = l.step(pipeline, srcRow, new Type.RelationType(withSlots));
        }
        Type.RelationType fullRow = new Type.RelationType(withSlots);
        ExprType rowInfo = new ExprType(fullRow, one);
        Map<String, TypedSpec> bindings = new LinkedHashMap<>();
        for (Type.Column c : cols) {
            bindings.put(c.name(), new TypedPropertyAccess(
                    new TypedVariable(rowVar, rowInfo), c.name(),
                    new ExprType(c.type(), c.multiplicity())));
        }
        for (Lift l : lifts) {
            bindings.put(l.property(), new TypedPropertyAccess(
                    new TypedVariable(rowVar, rowInfo), l.property(),
                    new ExprType(l.targetType(), one)));
        }
        ClassSource out = new ClassSource(mappingFqn, classFqn, ClassSource.UNION_SET_ID,
                pipeline, rowVar, bindings, fullRow);
        unionLeaves.put(out, leaves);
        return out;
    }

    /** One lifted navigation of a union: the property, the target class, the
     * source-side columns every arm projects for it (one per leaf read), the
     * routes (every leaf's, keys re-minted per leaf) and the predicate over
     * (union row, routed target row). */
    private record Lift(String property, Type targetType, TypedSpec target,
            List<Type.Column> srcCols, Map<String, String> srcPathByCol,
            List<com.legend.compiler.spec.typed.TypedNavigate.Route> routes,
            java.util.function.Function<Type.RelationType, TypedLambda> predicate) {
        @com.legend.Nullable String srcPathOf(int leaf, String col) {
            return srcPathByCol.get(leaf + " " + col);
        }
        TypedSpec step(TypedSpec source, Type.RelationType srcRow, Type.RelationType outRow) {
            var one = com.legend.compiler.element.type.Multiplicity.Bounded.ONE;
            return new com.legend.compiler.spec.typed.TypedNavigate(source, Optional.of(property),
                    target, predicate.apply(srcRow),
                    Optional.empty(), null, com.legend.compiler.spec.typed.TypedNavigate.Form.PRE_MAP,
                    new ExprType(Type.relation(outRow), one), routes);
        }
    }

    /** The lift of class-typed {@code p} over the leaves: null when some
     * leaf's function does not navigate it (loud at demand, as before). */
    private @com.legend.Nullable Lift liftOf(com.legend.compiler.element.Property p,
            List<ClassSource> leaves, List<TypedSpec> pipes) {
        var optional = com.legend.compiler.element.type.Multiplicity.Bounded.ZERO_ONE;
        List<com.legend.compiler.spec.typed.TypedNavigate> steps = new ArrayList<>();
        for (ClassSource leaf : leaves) {
            var st = Pipelines.navSteps(leaf.pipeline()).get(p.name());
            if (st == null || !(st.target() instanceof com.legend.compiler.spec.typed.TypedGetAll)
                    || st.form() != com.legend.compiler.spec.typed.TypedNavigate.Form.PRE_MAP) {
                return null;
            }
            steps.add(st);
        }
        Type targetType = Type.asClassType(p.type());
        List<Type.Column> srcCols = new ArrayList<>();
        Map<String, String> srcPathByCol = new LinkedHashMap<>();
        List<com.legend.compiler.spec.typed.TypedNavigate.Route> routes = new ArrayList<>();
        List<TypedSpec> preds = new ArrayList<>();
        List<String> sParams = new ArrayList<>();
        List<String> tParams = new ArrayList<>();
        List<Type.Column> keyCols = new ArrayList<>();
        java.util.Set<String> seenKeys = new LinkedHashSet<>();
        for (int i = 0; i < leaves.size(); i++) {
            var st = steps.get(i);
            TypedLambda pred = st.predicate();
            String sv = pred.parameters().get(0);
            String tv = pred.parameters().get(1);
            TypedSpec body = pred.body().get(pred.body().size() - 1);
            Type.RelationType leafRow = Type.requireRelationSchema(pipes.get(i).info().type());
            // source side: every read of the leaf's row becomes a column the
            // arm projects, NULL in the other arms
            List<String> sReads = new ArrayList<>();
            NavRewrites.collectReads(body, sv, sReads);
            Map<String, String> sRename = new LinkedHashMap<>();
            for (int k = 0; k < sReads.size(); k++) {
                String name = "__s_" + p.name() + "_" + i + "_" + k;
                Type t = pathType(leafRow, sReads.get(k));
                if (t == null) {
                    return null;    // the leaf's step reads beyond its row: not liftable here
                }
                srcCols.add(new Type.Column(name, t, optional));
                srcPathByCol.put(i + " " + name, sReads.get(k));
                sRename.put(sReads.get(k), name);
            }
            // target side: the leaf's routes (keys re-minted per leaf), or one
            // route of the class extent reading the leaf's raw target reads
            Map<String, String> tRename = new LinkedHashMap<>();
            if (!st.routes().isEmpty()) {
                Type.RelationType tRow = Type.requireRelationSchema(
                        pred.info().type() instanceof Type.FunctionType ft
                                ? ft.parameters().get(1).type() : null);
                for (var r : st.routes()) {
                    List<String> keys = new ArrayList<>();
                    for (String k : r.keyNames()) {
                        String nk = "__l" + i + k;
                        keys.add(nk);
                        tRename.put(k, nk);
                        if (seenKeys.add(nk)) {
                            Type.Column kc = tRow == null ? null
                                    : tRow.columns().stream().filter(c -> c.name().equals(k)).findFirst().orElse(null);
                            keyCols.add(new Type.Column(nk, kc == null ? Type.Primitive.ANY : kc.type(), optional));
                        }
                    }
                    routes.add(new com.legend.compiler.spec.typed.TypedNavigate.Route(
                            r.target(), r.rows(), r.cond(), r.targetReads(), keys));
                }
            } else {
                List<String> tReads = new ArrayList<>();
                NavRewrites.collectReads(body, tv, tReads);
                Type.RelationType tRow = Type.requireRelationSchema(
                        pred.info().type() instanceof Type.FunctionType ft
                                ? ft.parameters().get(1).type() : null);
                List<String> keys = new ArrayList<>();
                for (int k = 0; k < tReads.size(); k++) {
                    String nk = "__l" + i + "_" + p.name() + "_" + k;
                    keys.add(nk);
                    tRename.put(tReads.get(k), nk);
                    Type kt = tRow == null ? null : pathType(tRow, tReads.get(k));
                    keyCols.add(new Type.Column(nk, kt == null ? Type.Primitive.ANY : kt, optional));
                }
                routes.add(new com.legend.compiler.spec.typed.TypedNavigate.Route(
                        st.target(), leaves.get(i).pipeline(), pred, tReads, keys));
            }
            preds.add(NavRewrites.renameReads(NavRewrites.renameReads(body, sv, sRename), tv, tRename));
            sParams.add(sv);
            tParams.add(tv);
        }
        TypedSpec target = steps.get(0).target();
        final List<TypedSpec> fPreds = preds;
        final Type.RelationType tRowAll = new Type.RelationType(keyCols);
        java.util.function.Function<Type.RelationType, TypedLambda> predicate = srcRow -> {
            var one = com.legend.compiler.element.type.Multiplicity.Bounded.ONE;
            var boolOne = new ExprType(Type.Primitive.BOOLEAN, one);
            TypedVariable s = new TypedVariable("s", new ExprType(srcRow, one));
            TypedVariable t = new TypedVariable("t", new ExprType(tRowAll, one));
            TypedSpec or = null;
            for (int i = 0; i < fPreds.size(); i++) {
                TypedSpec re = NavRewrites.renameVar(NavRewrites.renameVar(fPreds.get(i),
                        sParams.get(i), "s", s.info()), tParams.get(i), "t", t.info());
                or = or == null ? re : new TypedNativeCall(orFn(), List.of(or, re), boolOne, null);
            }
            return new TypedLambda(List.of("s", "t"), List.of(java.util.Objects.requireNonNull(or)),
                    new ExprType(new Type.FunctionType(
                            List.of(new Type.Param(srcRow, one), new Type.Param(tRowAll, one)),
                            new Type.Param(Type.Primitive.BOOLEAN, one)), one));
        };
        return new Lift(p.name(), targetType, target, srcCols, srcPathByCol, routes, predicate);
    }

    private com.legend.compiler.element.TypedFunction orFn() {
        var fns = ctx.findFunction("meta::pure::functions::boolean::or")
                .stream().filter(f -> f.parameters().size() == 2).toList();
        if (fns.size() != 1) {
            throw new IllegalStateException("resolver bug: expected one 2-arg boolean::or");
        }
        return fns.get(0);
    }
