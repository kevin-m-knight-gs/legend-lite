// The measurement program (not committed as code: a manual legend_java_binary over //core:core_tests_lib, main class
// com.legend.lowering.ScratchScale, jvm_flags -Xss64m -Xmx12g). Its numbers are in measurements.txt beside it.
package com.legend.lowering;

import com.legend.Compiler;
import com.legend.compiler.element.ModelContext;
import com.legend.sql.SqlQuery;
import com.legend.sql.dialect.DuckDb;
import com.legend.testcases.SqlIslandCases;
import java.util.List;

public final class ScratchScale {
    static final String EMP = SqlIslandCases.SOURCE;

    public static void main(String[] args) throws Exception {
        String model = SqlIslandCases.model("type: DuckDB;\n specification: DuckDB { };\n auth: Test;");
        ModelContext ctx = Compiler.compileModel(model);
        String which = args.length > 0 ? args[0] : "both";
        if (which.equals("nest") || which.equals("both")) {
            for (int n : new int[] {5, 10, 15, 18, 20}) {
                String sql = "SELECT age AS x FROM " + EMP;
                for (int i = 1; i <= n; i++) {
                    sql = "SELECT x * x AS x FROM (" + sql + ") q" + i;
                }
                measure("nest", n, ctx, sql);
            }
        }
        if (which.equals("chain") || which.equals("both")) {
            for (int n : new int[] {25, 50, 100, 150, 200, 250}) {
                StringBuilder sql = new StringBuilder("SELECT t1.ID AS k FROM " + EMP + " t1");
                for (int i = 2; i <= n; i++) {
                    sql.append(" JOIN ").append(EMP).append(" t").append(i).append(" ON t").append(i)
                            .append(".ID = t").append(i - 1).append(".ID");
                }
                measure("chain", n, ctx, sql.toString());
            }
            for (int n : new int[] {25, 50, 100}) {
                StringBuilder sql = new StringBuilder("SELECT t1.ID AS k FROM " + EMP + " t1");
                for (int i = 2; i <= n; i++) {
                    sql.append(", ").append(EMP).append(" t").append(i);
                }
                sql.append(" WHERE t1.ID = t2.ID");
                for (int i = 3; i <= n; i++) {
                    sql.append(" AND t").append(i - 1).append(".ID = t").append(i).append(".ID");
                }
                measure("comma", n, ctx, sql.toString());
            }
        }
    }

    static void measure(String shape, int n, ModelContext ctx, String sql) {
        long t0 = System.nanoTime();
        String out;
        try {
            var parsed = com.legend.parser.SpecParser.parse("|#SQL{" + sql + "}#",
                    com.legend.parser.Dialect.LEGEND_LITE);
            List<com.legend.protocol.spec.ValueSpecification> statements =
                    parsed instanceof com.legend.protocol.spec.LambdaFunction lf && lf.parameters().isEmpty()
                            ? lf.body() : List.of(parsed);
            var resolved = Compiler.resolveQuery(statements, new com.legend.model.ImportScope(List.of()), ctx);
            SqlQuery q = Compiler.lowerResolved(resolved, ctx, SqlIslandCases.RUNTIME, false);
            long t1 = System.nanoTime();
            String rendered = new DuckDb().render(q);
            out = String.format(java.util.Locale.ROOT, "%s %d: plan %d ms, render %d ms, %d chars", shape, n,
                    (t1 - t0) / 1_000_000, (System.nanoTime() - t1) / 1_000_000, rendered.length());
        } catch (Throwable e) {
            out = String.format(java.util.Locale.ROOT, "%s %d: %s after %d ms: %s", shape, n,
                    e.getClass().getSimpleName(), (System.nanoTime() - t0) / 1_000_000,
                    String.valueOf(e.getMessage()).lines().findFirst().orElse(""));
        }
        System.out.println(out);
        System.out.flush();
    }
}
